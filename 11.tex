\begin{algorithm}[H]
\caption{Euclid's algorithm}
TWO non- negative integers $a$ and $b$
{$\gcd(a,b) $}\\
\While{$b \neq 0$}{
$ r \gets ab $ \;
$ a \gets b $ \;
$ b \gets r $\;
}
\Return $a$ \;
\end{algorithm}
